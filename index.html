      const total = properties.length;

      const sales = properties.filter(p => p.status === 'Sale').length;

      const rents = properties.filter(p => p.status === 'Rent').length;


      let saleablePrices = properties

        .filter(p => p.status === 'Sale' && p.saleableArea > 0)

        .map(p => (p.price * 1000000) / p.saleableArea);

      

      let avgPrice = saleablePrices.length > 0 

        ? Math.round(saleablePrices.reduce((a, b) => a + b, 0) / saleablePrices.length) 

        : 0;


      document.getElementById('stat-total').innerText = total;

      document.getElementById('stat-sale').innerText = sales;

      document.getElementById('stat-rent').innerText = rents;

      document.getElementById('stat-avgprice').innerText = '$' + avgPrice.toLocaleString();

    }

    // CALCULATE RATES

    function calculateRates() {

      const price = parseFloat(document.getElementById('edit-price').value) || 0;

      const gross = parseFloat(document.getElementById('edit-grossArea').value) || 0;

      const saleable = parseFloat(document.getElementById('edit-saleableArea').value) || 0;


      if (gross > 0 && saleable > 0) {

        document.getElementById('edit-efficiency').value = Math.round((saleable / gross) * 100) + ' %';

      } else {

        document.getElementById('edit-efficiency').value = '-- %';

      }


      if (price > 0 && saleable > 0) {

        document.getElementById('edit-saleablePrice').value = '$' + Math.round((price * 1000000) / saleable).toLocaleString();

      } else {

        document.getElementById('edit-saleablePrice').value = '$--';

      }

    }

    // SAVE PROPERTY TO SUPABASE

    async function saveProperty(e) {

      e.preventDefault();

      const id = document.getElementById('edit-id').value || "SK-" + Date.now();

      

      const photoElements = document.querySelectorAll('.grid-photo-url');

      const photosList = Array.from(photoElements).map(el => el.value);


      const propertyPayload = {

        id: id,

        ref: document.getElementById('edit-ref').value,

        status: document.getElementById('edit-status').value,

        district: document.getElementById('edit-district').value,

        type: document.getElementById('edit-type').value,

        title_zh: document.getElementById('edit-titleZh').value,

        title_en: document.getElementById('edit-titleEn').value,

        block: document.getElementById('edit-block').value,

        floor: document.getElementById('edit-floor').value,

        unit: document.getElementById('edit-unit').value,

        price: parseFloat(document.getElementById('edit-price').value),

        gross_area: parseInt(document.getElementById('edit-grossArea').value) || 0,

        saleable_area: parseInt(document.getElementById('edit-saleableArea').value),

        view: document.getElementById('edit-view').value,

        layout: document.getElementById('edit-layout').value,

        mgmt: document.getElementById('edit-mgmt').value,

        contact_name: document.getElementById('edit-contactName').value,

        contact_phone: document.getElementById('edit-contactPhone').value,

        notes: document.getElementById('edit-notes').value,

        photos: photosList.length > 0 ? photosList : ["https://images.unsplash.com/photo-1568605117036-5fe5e7bab0b7?w=800&auto=format&fit=crop&q=60"]

      };


      const { error } = await supabase.from('properties').upsert(propertyPayload);


      if (error) {

        alert("儲存失敗 (Save failed): " + error.message);

        return;

      }


      alert("樓盤資料已成功儲存並同步至雲端！");

      await fetchPropertiesFromCloud();

      switchTab('index');

    }

    function handleSimulatedUpload(event) {

      renderPhotoGrid(currentList);

    }

    // MORTGAGE CALCULATOR

    function runMortgageCalc() {

      const price = parseFloat(document.getElementById('calc-price').value) || 0;

      const ltvPercent = parseFloat(document.getElementById('calc-ltv').value) / 100;

      const years = parseInt(document.getElementById('calc-years').value);

      const ratePercent = parseFloat(document.getElementById('calc-rate').value) / 100;


      const loanAmount = price * ltvPercent;

      const downPayment = price - loanAmount;

      const monthlyRate = ratePercent / 12;

      const totalMonths = years * 12;

      

      let monthlyRepayment = 0;

      if (monthlyRate > 0 && totalMonths > 0) {

        monthlyRepayment = loanAmount * (monthlyRate * Math.pow(1 + monthlyRate, totalMonths)) / (Math.pow(1 + monthlyRate, totalMonths) - 1);

      }


      document.getElementById('res-downpayment').innerText = '$' + Math.round(downPayment).toLocaleString();

      document.getElementById('res-loanamount').innerText = '$' + Math.round(loanAmount).toLocaleString();

      document.getElementById('res-monthly').innerText = '$' + Math.round(monthlyRepayment).toLocaleString();


      let stampDuty = 0;

      if (price <= 3000000) stampDuty = 100;

      else if (price <= 3528240) stampDuty = 100 + (price - 3000000) * 0.1;

      else if (price <= 4500000) stampDuty = price * 0.015;

      else if (price <= 5315630) stampDuty = 67500 + (price - 4500000) * 0.1;

      else if (price <= 6000000) stampDuty = price * 0.0225;

      else if (price <= 7528240) stampDuty = 135000 + (price - 6000000) * 0.1;

      else if (price <= 9000000) stampDuty = price * 0.03;

      else if (price <= 10890580) stampDuty = 270000 + (price - 9000000) * 0.1;

      else if (price <= 20000000) stampDuty = price * 0.0375;

      else stampDuty = price * 0.0425;


      document.getElementById('res-stampduty').innerText = '$' + Math.round(stampDuty).toLocaleString();

    }
